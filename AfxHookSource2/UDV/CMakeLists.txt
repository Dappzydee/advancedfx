cmake_minimum_required(VERSION 3.24)
project(udv_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(udv_core Core.cpp Worker.cpp Compute.cpp CollisionScene.cpp)
set(UDV_COMPUTE_TARGET udv_core)
include(Cuda.cmake)
target_link_libraries(udv_core PUBLIC Threads::Threads)
if(NOT MSVC)
  target_compile_options(udv_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(udv_tests Tests.cpp)
target_link_libraries(udv_tests PRIVATE udv_core)
add_test(NAME udv_core COMMAND udv_tests)
add_executable(udv_benchmark Benchmark.cpp)
target_link_libraries(udv_benchmark PRIVATE udv_core)
add_executable(udv_gpu_validate GpuValidate.cpp)
target_link_libraries(udv_gpu_validate PRIVATE udv_core)
add_test(NAME udv_gpu_parity COMMAND udv_gpu_validate)
set_tests_properties(udv_gpu_parity PROPERTIES SKIP_RETURN_CODE 77)
add_executable(udv_collision_tests CollisionTests.cpp)
target_link_libraries(udv_collision_tests PRIVATE udv_core)
add_test(NAME udv_collision_conversion COMMAND udv_collision_tests)
